cmake_minimum_required(VERSION 3.22)
project(ClipPocket VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../JUCE" CACHE PATH "JUCE 8.0.4 source directory")
option(CLIP_BUILD_PLUGIN "Build JUCE plug-ins" ON)
option(CLIP_BUILD_UI_TESTS "Build processor/UI integration checks" OFF)
option(CLIP_ENABLE_OPENGL "Optional editor OpenGL (off at runtime by default)" ON)
set(CLIP_SANITIZER "none" CACHE STRING "none, address, undefined")
if(NOT MSVC AND NOT CLIP_SANITIZER STREQUAL "none")
 if(CLIP_SANITIZER STREQUAL "address")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
 elseif(CLIP_SANITIZER STREQUAL "undefined")
  add_compile_options(-fsanitize=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=undefined)
 endif()
endif()
enable_testing()
add_executable(ClipDSPTest Tests/dsp_test.cpp)
add_test(NAME ClipDSP COMMAND ClipDSPTest)
add_executable(ClipMeasurements Tests/measurements.cpp)
if(NOT CLIP_BUILD_PLUGIN)
 return()
endif()
if(WIN32)
 include(cmake/ApplyWindowsVBlankFix.cmake)
 clip_apply_windows_vblank_fix("${JUCE_DIR}")
endif()
add_subdirectory("${JUCE_DIR}" JUCE)
set(CLIP_FORMATS "VST3;AAX" CACHE STRING "Plug-in formats (VST3;AAX by default)")
# JUCE 8.0.4 includes the AAX SDK used by the Duck Pocket builds.
# A custom SDK remains optional; JUCE selects its bundled SDK otherwise.
if("AAX" IN_LIST CLIP_FORMATS AND AAX_SDK_PATH)
 juce_set_aax_sdk_path("${AAX_SDK_PATH}")
endif()
juce_add_plugin(ClipPocket
 COMPANY_NAME "Rainline Music" COMPANY_EMAIL "rainlinemusic@gmail.com"
 BUNDLE_ID "ru.rainlinemusic.clippocket" AAX_IDENTIFIER "ru.rainlinemusic.clippocket"
 PLUGIN_MANUFACTURER_CODE RnLn PLUGIN_CODE ClPk FORMATS ${CLIP_FORMATS}
 PRODUCT_NAME "Clip Pocket" DESCRIPTION "Perceptual clipper with selective kick protection"
 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
 EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE
 VST3_CATEGORIES Fx Dynamics AAX_CATEGORY Dynamics)
juce_generate_juce_header(ClipPocket)
target_sources(ClipPocket PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h Source/ClipDSP.h Source/UIStyle.h Source/SoftwareGlow.h)
target_compile_definitions(ClipPocket PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 CLIP_ENABLE_OPENGL=$<BOOL:${CLIP_ENABLE_OPENGL}> $<$<BOOL:${WIN32}>:NOMINMAX=1> $<$<BOOL:${WIN32}>:WIN32_LEAN_AND_MEAN=1>)
if(MSVC)
 target_compile_options(ClipPocket PRIVATE /utf-8)
else()
 target_compile_options(ClipPocket PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
target_link_libraries(ClipPocket PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(CLIP_ENABLE_OPENGL)
 target_link_libraries(ClipPocket PRIVATE juce::juce_opengl)
endif()
juce_add_binary_data(ClipAssets SOURCES Assets/Fonts/Inter-Regular.ttf Assets/Fonts/Inter-Medium.ttf Assets/Designer/DialLarge.png Assets/Designer/DialSmall.png)
set_target_properties(ClipAssets PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
target_link_libraries(ClipPocket PUBLIC ClipAssets)
if(CLIP_BUILD_UI_TESTS)
 juce_add_console_app(ClipIntegrationTest PRODUCT_NAME "Clip Pocket integration checks")
 juce_generate_juce_header(ClipIntegrationTest)
 target_sources(ClipIntegrationTest PRIVATE Tests/integration_test.cpp)
 target_include_directories(ClipIntegrationTest PRIVATE Source "${CMAKE_CURRENT_BINARY_DIR}/ClipPocket_artefacts/JuceLibraryCode")
 target_compile_definitions(ClipIntegrationTest PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(ClipIntegrationTest PRIVATE ClipPocket juce::juce_audio_utils)
 add_test(NAME ClipIntegration COMMAND ClipIntegrationTest "${CMAKE_CURRENT_BINARY_DIR}/ui-captures")
endif()
foreach(format VST3 AAX AU)
 if(TARGET ClipPocket_${format})
  add_custom_command(TARGET ClipPocket_${format} POST_BUILD
   COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:ClipPocket_${format}>/../Resources"
   COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "$<TARGET_FILE_DIR:ClipPocket_${format}>/../Resources/Inter-OFL.txt" VERBATIM)
 endif()
endforeach()

if(TARGET ClipPocket_Standalone)
 if(APPLE)
  set(clip_standalone_license_dir "$<TARGET_FILE_DIR:ClipPocket_Standalone>/../Resources")
 else()
  set(clip_standalone_license_dir "$<TARGET_FILE_DIR:ClipPocket_Standalone>")
 endif()
 add_custom_command(TARGET ClipPocket_Standalone POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${clip_standalone_license_dir}"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "${clip_standalone_license_dir}/Inter-OFL.txt" VERBATIM)
endif()
